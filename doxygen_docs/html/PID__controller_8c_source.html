107double headingError(double target, double current) {
108 double error = target - current;
109 while (error > 180.0) error -= 360.0;
110 while (error < -180.0) error += 360.0;
111 return error;
112}
113
121double pidCompute(double error, double dt) {
122 integral += error * dt;
123 double derivative = (error - prevError) / dt;
124 prevError = error;
125
126 double output = (KP * error) + (KI * integral) + (KD * derivative);
127
128 // Normalize output range (e.g., -1 to 1)
129 if (output > 255) output = 255;
130 if (output < -255) output = -255;
131
132 return output;
133}

158void PID_trigger(){
159
160 // Get latest heading and course to calculate error
161 getlatestHeading(&currentHeading);
162 LOS_getdesiredheading(&desiredHeading);
163
164 double dt;
165 TickType_t now = xTaskGetTickCount();
166 dt = (now - lastTime) / 1000.0; // convert to seconds
167 lastTime = now;
168
169 double error = headingError(desiredHeading, currentHeading);
170 double steering = pidCompute(error, dt);
171
172 double leftSpeed = BASE_SPEED + steering;
173 double rightSpeed = BASE_SPEED - steering;
174
175 // If the desired action is effectively zero (no base speed and very
176 // small heading error), keep motors off so they don't waste power.
177 const double EPS_ERROR = 0.5; // degrees (tolerance for considering error zero)
178 bool keepMotorsOff = (fabs(error) < EPS_ERROR) && (BASE_SPEED == 0);
179
180 // Clamp to max limits first
181 if (leftSpeed > MAX_SPEED) leftSpeed = MAX_SPEED;
182 if (leftSpeed < -MAX_SPEED) leftSpeed = -MAX_SPEED;
183 if (rightSpeed > MAX_SPEED) rightSpeed = MAX_SPEED;
184 if (rightSpeed < -MAX_SPEED) rightSpeed = -MAX_SPEED;
185
186 // Apply minimum effective speed to overcome friction when a non-zero
187 // command is requested. Preserve exact zero when keepMotorsOff is true.
188 if (!keepMotorsOff) {
189 if ((leftSpeed > 0) && (leftSpeed < MIN_SPEED)) leftSpeed = MIN_SPEED;
190 if ((leftSpeed < 0) && (leftSpeed > -MIN_SPEED)) leftSpeed = -MIN_SPEED;
191
192 if ((rightSpeed > 0) && (rightSpeed < MIN_SPEED)) rightSpeed = MIN_SPEED;
193 if ((rightSpeed < 0) && (rightSpeed > -MIN_SPEED)) rightSpeed = -MIN_SPEED;
194 } else {
195 leftSpeed = 0;
196 rightSpeed = 0;
197 }
198
199 Motor_Set_Speed((int16_t)leftSpeed, (int16_t)rightSpeed);
200
201 #ifdef DEBUG_PID_CONTROLLER
202 char msg[128];
203 sprintf(msg, "PID Debug: DH=%.2f CH=%.2f Err=%.2f LSpd=%.2f RSpd=%.2f\r\n",
204 desiredHeading, currentHeading, error, leftSpeed, rightSpeed);
205 UART_puts(msg);
206 #endif
207}

214void PID_Controller(void *argument)
215{
216 osDelay(200); // wait a second to make sure everything is started
217
218 while (1)
219 {
220 // Non-blocking: Try to read a key from the queue. If none available, continue doing other work.
221
222 LOS_getwaypointhold_status(&pid_waypoint_hold);
223
224 if (hKeyPID_Queue != NULL)
225 {
226 if (xQueueReceive(hKeyPID_Queue, &key, 0) == pdTRUE)
227 {
228 // Process key
229 switch(key)
230 {
231 case 5:
232 UART_puts("\r\n Toggle received in PID_Controller\r\n");
233 // Toggle PID controller state and inform Route Performer with same state
234 enablePID = !enablePID;
235 // set_RP_algoState(enablePID);
236 set_LOS_algoState(enablePID);
237
238 if (enablePID)
239 UART_puts("PID Controller enabled\r\n");
240 else
241 UART_puts("PID Controller disabled\r\n");
242 break;
243 case 6:
244 UART_puts("\r\n Obstacle avoidance toggle received in PID_Controller\r\n");
245 // Toggle obstacle avoidance state
246 enableObstacleAvoidance = !enableObstacleAvoidance;
247
248 if (enableObstacleAvoidance)
249 UART_puts("Obstacle avoidance enabled\r\n");
250 else
251 UART_puts("Obstacle avoidance disabled\r\n");
252 break;
253 default:
254 UART_puts("\r\nInvalid key pressed for PID_Controller\r\n");
255 break; // continue loop
256 }
257 }
258 }
259
260 if (enablePID && !pid_waypoint_hold && !obstacleFlag)
261 {
262 PID_trigger();
263 osDelay(10); // Control loop delay
264
265 }
266 else if(!enablePID)
267 {
268 Motor_Set_Speed(0, 0); // Stop motors when PID is disabled
269 }
270 else if(pid_waypoint_hold)
271 {
272 Motor_Set_Speed(0, 0); // Stop motors when in waypoint hold
273 }
274 else if(obstacleFlag && enableObstacleAvoidance)
275 {
276 obstacleAvoidance();
277 }
278
279 osDelay(10); // Idle delay when PID is disabled
280 }
281}
