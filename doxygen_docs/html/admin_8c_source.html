337void CreateTasks(void)
338{
339 PTASKDATA ptd = tasks; // tasks == &tasks[0]: both are addresses of first struct
340 vTaskSuspendAll(); // zet de scheduler stil en dwing FreeRTOS om eerst alle taken te starten
341
342 for (; ptd->func != NULL; ptd++) // loopje zonder teller; gewoon de pointer steeds ophogen...
343 {
344 UART_puts("\r\nCreated: ");
345 UART_puts((char *)ptd->attr.name);
346
347 // nb. kijk naar de haakjes van argument 3: het gaat om het adres van (structure attr).
348 ptd->hTask = osThreadNew(ptd->func, ptd->argument, &(ptd->attr));
349 }
350
351 UART_puts("\r\n");
352 xTaskResumeAll(); // start nu de scheduler: play ball
353 DisplayTaskData(); // display alle taskdata op UART
354}

413void StartStopTask(int task_id)
414{
415 PTASKDATA ptd = tasks;
416 unsigned int task_nr;
417 TaskStatus_t xTaskDetails;
418
419 for (task_nr = 1; ptd->func != NULL; ptd++, task_nr++)
420 if (task_nr == task_id) // hebbes
421 {
422 vTaskGetInfo (ptd->hTask, &xTaskDetails, pdTRUE, eInvalid);
423 UART_puts("\r\n");
424 UART_puts(ptd->attr.name);
425
426 // een task kan meerdere statussen hebben; het wordt simpeler
427 // als je alleen de eSuspended checkt.
428 if (xTaskDetails.eCurrentState == eSuspended)
429 {
430 vTaskResume(ptd->hTask);
431 UART_puts(" resumed.");
432 }
433 else // bv. eBlocked, eRunning, eInvalid etc.
434 {
435 vTaskSuspend(ptd->hTask);
436 UART_puts(" suspended.");
437 }
438 return;
439 }

447void DisplayTaskData(void)
448{
449 PTASKDATA ptd = tasks;
450 UBaseType_t highwatermark;
451 TaskStatus_t xTaskDetails;
452 unsigned int free, totalalloc = 0;
453
454 unsigned int task_nr; // tasknr for changing priority
455
456 for (task_nr=1; ptd->func != NULL; ptd++, task_nr++)
457 {
458 highwatermark = uxTaskGetStackHighWaterMark(ptd->hTask); // amount of free bytes
459 free = (highwatermark * 100) / ptd->attr.stack_size; // in percentage
460 totalalloc += ptd->attr.stack_size;
461
462 vTaskGetInfo (ptd->hTask, &xTaskDetails, pdTRUE, eInvalid); // get task status
463
464 UART_puts("\r\n\t["); UART_putint(task_nr); UART_puts("] ");
465 UART_puts(ptd->attr.name);
466 UART_puts("\t priority: "); UART_putint(ptd->attr.priority);
467 UART_puts("\t stacksize: "); UART_putint(ptd->attr.stack_size * 4);
468 UART_puts("\t free: "); UART_putint(highwatermark*4);
469 UART_puts("\t used: "); UART_putint(100 - free); UART_puts("%");
470 UART_puts("\t status: "); UART_puts(xTaskDetails.eCurrentState == eSuspended ? "suspended": "running");
471 }
472 UART_puts("\r\n\tTotal heap: "); UART_putint(configTOTAL_HEAP_SIZE);
473 UART_puts(" Allocated task stack: "); UART_putint(totalalloc * 4);
474 UART_puts(" Free heap space: "); UART_putint(configTOTAL_HEAP_SIZE - (totalalloc * 4));
475 UART_puts("\r\n");
476}
