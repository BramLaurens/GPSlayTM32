347void CreateTasks(void)
348{
349 PTASKDATA ptd = tasks; // tasks == &tasks[0]: both are addresses of first struct
350 vTaskSuspendAll(); // zet de scheduler stil en dwing FreeRTOS om eerst alle taken te starten
351
352 for (; ptd->func != NULL; ptd++) // loopje zonder teller; gewoon de pointer steeds ophogen...
353 {
354 UART_puts("\r\nCreated: ");
355 UART_puts((char *)ptd->attr.name);
356
357 // nb. kijk naar de haakjes van argument 3: het gaat om het adres van (structure attr).
358 ptd->hTask = osThreadNew(ptd->func, ptd->argument, &(ptd->attr));
359 }
360
361 UART_puts("\r\n");
362 xTaskResumeAll(); // start nu de scheduler: play ball
363 DisplayTaskData(); // display alle taskdata op UART
364}

423void StartStopTask(int task_id)
424{
425 PTASKDATA ptd = tasks;
426 unsigned int task_nr;
427 TaskStatus_t xTaskDetails;
428
429 for (task_nr = 1; ptd->func != NULL; ptd++, task_nr++)
430 if (task_nr == task_id) // hebbes
431 {
432 vTaskGetInfo (ptd->hTask, &xTaskDetails, pdTRUE, eInvalid);
433 UART_puts("\r\n");
434 UART_puts(ptd->attr.name);
435
436 // een task kan meerdere statussen hebben; het wordt simpeler
437 // als je alleen de eSuspended checkt.
438 if (xTaskDetails.eCurrentState == eSuspended)
439 {
440 vTaskResume(ptd->hTask);
441 UART_puts(" resumed.");
442 }
443 else // bv. eBlocked, eRunning, eInvalid etc.
444 {
445 vTaskSuspend(ptd->hTask);
446 UART_puts(" suspended.");
447 }
448 return;
449 }

457void DisplayTaskData(void)
458{
459 PTASKDATA ptd = tasks;
460 UBaseType_t highwatermark;
461 TaskStatus_t xTaskDetails;
462 unsigned int free, totalalloc = 0;
463
464 unsigned int task_nr; // tasknr for changing priority
465
466 for (task_nr=1; ptd->func != NULL; ptd++, task_nr++)
467 {
468 highwatermark = uxTaskGetStackHighWaterMark(ptd->hTask); // amount of free bytes
469 free = (highwatermark * 100) / ptd->attr.stack_size; // in percentage
470 totalalloc += ptd->attr.stack_size;
471
472 vTaskGetInfo (ptd->hTask, &xTaskDetails, pdTRUE, eInvalid); // get task status
473
474 UART_puts("\r\n\t["); UART_putint(task_nr); UART_puts("] ");
475 UART_puts(ptd->attr.name);
476 UART_puts("\t priority: "); UART_putint(ptd->attr.priority);
477 UART_puts("\t stacksize: "); UART_putint(ptd->attr.stack_size * 4);
478 UART_puts("\t free: "); UART_putint(highwatermark*4);
479 UART_puts("\t used: "); UART_putint(100 - free); UART_puts("%");
480 UART_puts("\t status: "); UART_puts(xTaskDetails.eCurrentState == eSuspended ? "suspended": "running");
481 }
482 UART_puts("\r\n\tTotal heap: "); UART_putint(configTOTAL_HEAP_SIZE);
483 UART_puts(" Allocated task stack: "); UART_putint(totalalloc * 4);
484 UART_puts(" Free heap space: "); UART_putint(configTOTAL_HEAP_SIZE - (totalalloc * 4));
485 UART_puts("\r\n");
486}
