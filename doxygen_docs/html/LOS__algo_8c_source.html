125static void latlon_to_xy(double lat_ref, double lon_ref,
126 double lat, double lon,
127 double *x, double *y)
128{
129 double R = 6371000.0; // Earth radius in meters
130 double dLat = deg2rad(lat - lat_ref);
131 double dLon = deg2rad(lon - lon_ref);
132 double lat_ref_rad = deg2rad(lat_ref);
133 *x = R * dLon * cos(lat_ref_rad);
134 *y = R * dLat;
135}

146static double bearing_deg(double lat1, double lon1, double lat2, double lon2)
147{
148 double y = sin(deg2rad(lon2 - lon1)) * cos(deg2rad(lat2));
149 double x = cos(deg2rad(lat1)) * sin(deg2rad(lat2)) -
150 sin(deg2rad(lat1)) * cos(deg2rad(lat2)) * cos(deg2rad(lon2 - lon1));
151 double brng = atan2(y, x);
152 brng = rad2deg(brng);
153 if (brng < 0) brng += 360.0;
154 return brng;
155}

166static double distance_meters(double lat1, double lon1, double lat2, double lon2)
167{
168 const double R = 6371000.0; // Earth radius in meters
169 double dLat = deg2rad(lat2 - lat1);
170 double dLon = deg2rad(lon2 - lon1);
171
172 double a = sin(dLat / 2.0) * sin(dLat / 2.0) +
173 cos(deg2rad(lat1)) * cos(deg2rad(lat2)) *
174 sin(dLon / 2.0) * sin(dLon / 2.0);
175
176 double c = 2.0 * atan2(sqrt(a), sqrt(1.0 - a));
177
178 return R * c;
179}

188static double angle_diff(double a, double b)
189{
190 double diff = fmod(a - b + 540.0, 360.0) - 180.0;
191 return diff;

207GPS_Route *LOS_Navigate(
208 GPS_Route *prev_wp,
209 GPS_Route *curr_wp,
210 double gps_lat,
211 double gps_lon,
212 double heading_deg,
213 double ref_lat,
214 double ref_lon,
215 LOS_Output_t *los_out)
216{
217 // --- Check if route is finished ---
218 if (curr_wp == NULL) {
219 los_out->route_complete = true;
220 los_out->waypoint_reached = true;
221 los_out->desired_bearing = 0;
222 los_out->heading_error = 0;
223 los_out->distance_to_wp = 0;
224 return NULL;
225 }
226
227 // --- Compute distance to current target (B) ---
228 double dist_to_B = distance_meters(gps_lat, gps_lon, curr_wp->latitude, curr_wp->longitude);
229 los_out->distance_to_wp = dist_to_B;
230
231 // --- Check if we reached current waypoint ---
232 if (dist_to_B < ARRIVAL_RADIUS_M) {
233 los_out->waypoint_reached = true;
234
235 if (curr_wp->Next_point == NULL) {
236 los_out->route_complete = true;
237 return NULL; // End of route
238 }
239 // Advance to next waypoint
240 return curr_wp->Next_point;
241 }
242
243 los_out->waypoint_reached = false;
244 los_out->route_complete = false;
245
246 // --- If there's no previous waypoint (start of route), just go directly to B ---
247 if (prev_wp == NULL) {
248 los_out->desired_bearing = bearing_deg(gps_lat, gps_lon, curr_wp->latitude, curr_wp->longitude);
249 los_out->heading_error = angle_diff(los_out->desired_bearing, heading_deg);
250 return curr_wp;
251 }
252
253 // --- Convert A, B, R to local x,y coordinates ---
254 double Ax, Ay, Bx, By, Rx, Ry;
255 latlon_to_xy(ref_lat, ref_lon, prev_wp->latitude, prev_wp->longitude, &Ax, &Ay);
256 latlon_to_xy(ref_lat, ref_lon, curr_wp->latitude, curr_wp->longitude, &Bx, &By);
257 latlon_to_xy(ref_lat, ref_lon, gps_lat, gps_lon, &Rx, &Ry);
258
259 // --- Compute the line-of-sight (A->B vector) ---
260 double ABx = Bx - Ax;
261 double ABy = By - Ay;
262 double AB_len = sqrt(ABx*ABx + ABy*ABy);
263 if (AB_len < 1e-6) AB_len = 1e-6; // avoid divide by zero
264
265 // --- Project rover position R onto line A-B ---
266 double t = ((Rx - Ax)*ABx + (Ry - Ay)*ABy) / (AB_len*AB_len);
267
268 // --- Compute lookahead point T ---
269 double lookahead_t = t + (LOOKAHEAD_DISTANCE_M / AB_len);
270 if (lookahead_t > 1.0) lookahead_t = 1.0; // don’t go beyond B, clamping
271 double Tx = Ax + lookahead_t * ABx;
272 double Ty = Ay + lookahead_t * ABy;
273
274 // --- Compute bearing from R to T ---
275 double T_lat = ref_lat + (Ty / 6371000.0) * (180.0 / M_PI);
276 double T_lon = ref_lon + (Tx / (6371000.0 * cos(deg2rad(ref_lat)))) * (180.0 / M_PI);
277 los_out->desired_bearing = bearing_deg(gps_lat, gps_lon, T_lat, T_lon);
278
279 // --- Compute heading error ---
280 los_out->heading_error = angle_diff(los_out->desired_bearing, heading_deg);
281
282 return curr_wp;
283}

292void onGPSupdate(double gps_lat, double gps_lon, double heading_deg)
293{
294 // Get route head
295 if (curr_wp == NULL)
296 curr_wp = Route_Pointer_Request(); // start at first waypoint
297
298 current_LOS_waypoint_number = (curr_wp != NULL) ? curr_wp->nodeNumber : -1;
299
300 curr_wp = LOS_Navigate(
301 prev_wp,
302 curr_wp,
303 gps_lat,
304 gps_lon,
305 heading_deg,
306 ref_lat, // reference latitude (approx local)
307 ref_lon, // reference longitude
308 &los
309 );
310
311 if (los.route_complete) {
312 printf("Route complete!\n");
313 hold_LOS_at_waypoint = true; // Reset hold flag
314 return;
315 }
316
317 if (los.waypoint_reached) {
318 // Advance prev_wp pointer
319 hold_LOS_at_waypoint = true;
320 osDelay(2000); // Hold for 2 seconds at waypoint
321 hold_LOS_at_waypoint = false;
322 prev_wp = curr_wp;
323 // curr_wp already advanced inside LOS_Navigate
324 }
325
326 los_desiredheading = los.desired_bearing;
327
328 // Use los.desired_bearing and los.heading_error in PID
329 char buffer[50];
330 sprintf(buffer, "Desired LOS heading: %.1f", los.desired_bearing);
331 UART_puts(buffer);
332
333}

351void LOS_caller(void *argument)
352{
353 while(1)
354 {
355 if(enable_LOS_algo)
356 {
357 UART_puts("LOS waiting for notification \r\n");
358 ulTaskNotifyTake(pdTRUE, portMAX_DELAY); // Wait for GPS update notification
359 UART_puts("LOS notified by GPS task\r\n");
360
361 startRoute();
362 getlatest_GNRMC(&gnrmc_los);
363 double gps_lat = convert_decimal_degrees(gnrmc_los.latitude, &gnrmc_los.NS_ind);
364 double gps_lon = convert_decimal_degrees(gnrmc_los.longitude, &gnrmc_los.EW_ind);
365 double heading_deg;
366 getlatestHeading(&heading_deg);
367 onGPSupdate(gps_lat, gps_lon, heading_deg);
368
369 }
370 osDelay(1000); // Run every 1000 ms
371 }
