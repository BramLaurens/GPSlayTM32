166// Zet het meegegeven positief getal op het scherm
167void LCD_putint(unsigned int num)
168{
169 unsigned int rest;
170 static unsigned char chars[10] = "0123456789"; // possible chars
171 char s[17]; // display has 16 chars [0..15] + 1 closing null
172 s[16]='\0'; // closing null set
173 int i=15; // startposition at last display char
174
175 // kinda function itoa, integer to ascii
176 if(num==0)
177 s[i--]='0'; // '0' at 14
178 else
179 {
180 while(num>0 && i)
181 {
182 rest = num%10;
183 num /= 10;
184 s[i--]=chars[rest];
185 }
186 }
187 LCD_put(&s[i+1]); // first, set i to starting pos, undo one i-- too much
188}

190// Stuurt een 4-bits commando naar het display
191static void LCD_writenibble(unsigned char data)
192{
193 ClearBits();
194
195 if (data & 0x01)
196 HAL_GPIO_WritePin(LCD_D7, GPIO_PIN_SET);
197
198 if (data & 0x02)
199 HAL_GPIO_WritePin(LCD_D6, GPIO_PIN_SET);
200
201 if (data & 0x04)
202 HAL_GPIO_WritePin(LCD_D5, GPIO_PIN_SET);
203
204 if (data & 0x08)
205 HAL_GPIO_WritePin(LCD_D4, GPIO_PIN_SET);
206
207}

210// Stuurt een 8-bits commando naar het display
211static void LCD_writebyte(unsigned char data)
212{
213 /* hoogste 4 bits */
214 HAL_GPIO_WritePin(LCD_EN, GPIO_PIN_SET);
215 LCD_writenibble((data>>4)&0x0F);
216 HAL_GPIO_WritePin(LCD_EN, GPIO_PIN_RESET);
217
218 HAL_Delay(2);
219
220 /* laagste 4 bits */
221 HAL_GPIO_WritePin(LCD_EN, GPIO_PIN_SET);
222 LCD_writenibble(data&0x0F);
223 HAL_GPIO_WritePin(LCD_EN, GPIO_PIN_RESET);
224
225 HAL_Delay(2);
226}
