49void PWM_Set(uint32_t freq, uint32_t dutyCycle, TIM_HandleTypeDef *htim, uint32_t channel)
50{
51 uint32_t timer_clk = HAL_RCC_GetSysClockFreq() / CLOCK_DIVISION; //Calculate timer clock by dividing system clock by 2 (APB1 prescaler)
52 uint32_t prsc = htim->Instance->PSC; //Get prescaler value from TIM12
53 uint32_t arr = (timer_clk / ((prsc + 1) * freq)) - 1; //Calculate auto-reload register value to achieve desired frequency
54 __HAL_TIM_SET_AUTORELOAD(htim, arr); //Set the auto-reload register with the calculated value
55
56 uint32_t duty = (arr + 1) * dutyCycle / MAX_DUTY_CYCLE; //Calculate compare value to achieve desired duty cycle
57 __HAL_TIM_SET_COMPARE(htim, channel, duty); //Set the compare register with the calculated value
58}
