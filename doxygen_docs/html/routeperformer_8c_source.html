133 GPS_Route *temp = pRoute_copy;
134
135 // Find the correct waypoint in the linked list

157double Calc_Angle(GPS_Route *pRoute)
158{
159 // Convert all positions to radians
160 double lat1 = dGPS_localcopy4.latitude * M_PI / 180.0;
161 double lon1 = dGPS_localcopy4.longitude * M_PI / 180.0;
162 double lat2 = pRoute->latitude * M_PI / 180.0;
163 double lon2 = pRoute->longitude * M_PI / 180.0;
164
165 double dLon = lon2 - lon1;
166
167 double y = sin(dLon) * cos(lat2);
168 double x = cos(lat1) * sin(lat2) - sin(lat1) * cos(lat2) * cos(dLon);
169 double angle = atan2(y, x) * 180.0 / M_PI;
170
171 if (angle < 0) angle += 360.0;
172
173 // --- Apply Utrecht magnetic declination (≈ +3.5° East) ---
174 double declination_deg = 3.5;
175 angle -= declination_deg; // subtract because declination is east-positive
176
177 if (angle < 0) angle += 360.0;
178 else if (angle >= 360) angle -= 360.0;
179
180 return angle;
181}

265double Distance_Till_Waypoint(int Working_routing_point)
266{
267 // UART_puts("\r\n Starting distance calculation \r \n");
268 GPS_Route *temp = pRoute_copy;
269
270 if(update_GPS_loc() < 0) // update the gps location and check for errors
271 {
272 return -1; // error updating gps location
273 }
274 //Check if the structs are valid
275 temp = check_structs(Working_routing_point);
276
277 if(temp == NULL){
278 UART_puts("Error one of the structs is not filled or correct (Distance_Till_Waypoint)\r \n");
279 return -1;
280 }
281
282
283 // Calculate the distance between the current location and the waypoint
284 double dLat = (dGPS_localcopy4.latitude - temp->latitude);
285 double dLong = (dGPS_localcopy4.longitude - temp->longitude);
286
287
288 // calc from lat, long to x, y using Rijksdriehoeks conversion
289 // Reference point (Amersfoort) for now this will change to our Ref gps
290 const double phi0 = 52.15517440; // Lat
291 const double lam0 = 5.38720621; // Long
292
293 double phi = (dGPS_localcopy4.latitude - phi0) * 0.36; //Lat
294 double lambda = (dGPS_localcopy4.longitude - lam0) * 0.36; // Long
295
296 // Base coordinates of amersfoort change this to our using amersfoort as ref or dont and see the distance in X, Y from amersfoort
297 const double x0 = 155000.0;
298 const double y0 = 463000.0;
299
300 // Compute RD X
301 double x_rd =
302 (190094.945 * lambda) +
303 (-11832.228 * phi * lambda) +
304 (-114.221 * phi * lambda * lambda) +
305 (-32.391 * pow(lambda, 3)) +
306 (-0.705 * phi) +
307 (-2.340 * phi * phi * lambda) +
308 (-0.608 * phi * phi) +
309 (-0.008 * lambda) +
310 (0.148 * pow(lambda, 4));
311
312 // Compute RD Y
313 double y_rd =
314 (309056.544 * phi) +
315 (3638.893 * lambda * lambda) +
316 (73.077 * phi * phi) +
317 (-157.984 * phi * lambda * lambda) +
318 (59.788 * pow(phi, 3)) +
319 (0.433 * lambda) +
320 (-6.439 * phi * phi * lambda) +
321 (-0.032 * lambda * lambda) +
322 (0.092 * pow(phi, 4)) +
323 (-0.054 * phi * pow(lambda, 3));
324
325 // Now calculate the distance between the two points in cartesian coordinates
326 double x = x0 + x_rd;
327 double y = y0 + y_rd;
328 double z = pow(x,2) + pow(y,2); // now in cartesian you can triangulate with pytharogrian theorem for you distance
329 double distance_local = sqrt(z);
330
331 // UART_puts("Distance calculated: ");
332 // char Buffer[100]; // buffer for sprintf for debugging the float
333 // sprintf(Buffer, "%2.4f \r \n", distance_local);
334 // UART_puts(Buffer);

346int Completed_waypoint(double Distance_to_point)
347{
348 if(update_GPS_loc() < 0) // update the gps location and check for errors
349 {
350 return -1; // error updating gps location

356 UART_puts("Error no data in waypoint structs received (Completed_waypoint) \r \n");
357 return -1;
358 }
359
360 // Check if close enough to the next waypoint
361 if(Distance_to_point < Error_marge_completed_waypoint ) // Check if the rover is close enough to the waypoint see header for #define
362 {
363 int Working_routing_point = Give_NodeNumber();
364
365 rp_waypoint_hold = true; // Pause at waypoint
366
367 UART_puts("Waypoint reached! Calculating next waypoint...\r \n");
368 UART_puts(" Current waypoint got: ");
369 UART_putint(Working_routing_point); // shows the nr of the waypoint got
370 UART_puts("\r \n");
371
372 GPS_Route *temp = check_structs(Working_routing_point); // get the correct struct
373
374 if(temp == NULL) {
375 UART_puts("Error: current waypoint not found in list (Completed_waypoint)\r \n");
376 return -1;
377 }
378
379 if((temp->Next_point == NULL)) // end of the list
380 {
381 UART_puts("Last waypoint achieved! \r \n");
382 return -2 ; // indicates last waypoint reached
383 }
384
385 UART_puts("Found current waypoint in linked list, returning next waypoint's nodeNumber \r \n");
386 /* Return the actual nodeNumber of the next waypoint instead of assuming sequential numbering */
387 return temp->Next_point->nodeNumber;
388 }
389 // Not close enough to the waypoint just return the current waypoint
390 return Give_NodeNumber();
391}

424

427

431
432 if(xSemaphoreTake(hAngle_Mutex, portMAX_DELAY) == pdTRUE) // Take mutex before updating shared angle variable

435 xSemaphoreGive(hAngle_Mutex); // Release mutex after updating

439 RP_wpCurrent = Working_routing_point; // Update current waypoint number for external use

441
442 // Wait at waypoint for a few seconds when reached
443 if(rp_waypoint_hold)
444 {
445 osDelay(4000);
446 rp_waypoint_hold = false;
