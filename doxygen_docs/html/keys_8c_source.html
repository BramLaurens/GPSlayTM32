33void KEYS_init(void)
34/* Keys initialize
35 * In this function the ports are set for the keyboard.
36 * The rows are outputs and the columns are input.
37 */
38{
39 __HAL_RCC_GPIOD_CLK_ENABLE();
40 GPIO_InitTypeDef gpio;
41 gpio.Mode = GPIO_MODE_OUTPUT_PP;
42 gpio.Pull = GPIO_NOPULL;
43 gpio.Speed = GPIO_SPEED_FREQ_VERY_HIGH;
44
45 gpio.Pin = KEY_ROW;
46 HAL_GPIO_Init(PKEY_ROW, &gpio);
47
48 gpio.Pin = KEY_COL;
49 gpio.Mode = GPIO_MODE_INPUT;
50 gpio.Pull = GPIO_PULLDOWN;
51 HAL_GPIO_Init(PKEY_COL, &gpio);
52 HAL_GPIO_WritePin(PKEY_ROW, KEY_ROW, 0);
53}

68unsigned int KEYS_read(void)
69/* Keys read
70 * This function reads which key is pressed.
71 * It does so by making a row high and check if there is a connection with a column.
72 * If there is a connection the value is returned.
73 */
74{
75 unsigned int key = 0;
76 unsigned out = 0;
77
78 HAL_GPIO_WritePin(PKEY_ROW, KEY_R1, 0);
79 HAL_GPIO_WritePin(PKEY_ROW, KEY_R4, 1);
80 key = KEYS_kolom();
81 if( key != 0 ) out= 12+key;
82
83 HAL_GPIO_WritePin(PKEY_ROW, KEY_R4, 0);
84 HAL_GPIO_WritePin(PKEY_ROW, KEY_R3, 1);
85 key = KEYS_kolom();
86 if( key != 0 ) out = 8+key;
87
88 HAL_GPIO_WritePin(PKEY_ROW, KEY_R3, 0);
89 HAL_GPIO_WritePin(PKEY_ROW, KEY_R2, 1);
90 key = KEYS_kolom();
91 if( key != 0 ) out = 4+key;
92
93 HAL_GPIO_WritePin(PKEY_ROW, KEY_R2, 0);
94 HAL_GPIO_WritePin(PKEY_ROW, KEY_R1, 1);
95 key = KEYS_kolom();
96 if( key != 0 ) out = key;
97
98 return out;
99}

101unsigned int KEYS_kolom(void)
102/* Keys Column
103 * In this function the columns are checked if there is a connection.
104 * If a column has a connection the return value is the column number.
105 */
106{
107 unsigned int key = 0;
108 if (HAL_GPIO_ReadPin(PKEY_COL, KEY_K1) == 1) key = 1;
109 if (HAL_GPIO_ReadPin(PKEY_COL, KEY_K2) == 1) key = 2;
110 if (HAL_GPIO_ReadPin(PKEY_COL, KEY_K3) == 1) key = 3;
111 if (HAL_GPIO_ReadPin(PKEY_COL, KEY_K4) == 1) key = 4;
112 return key;
113}
