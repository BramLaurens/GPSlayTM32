138char UART_get(void)
139{
140 unsigned char uart_char = -1;
141
142 if (__HAL_UART_GET_FLAG(&huart2, UART_FLAG_RXNE) != 0) // check for data available using HAL
143 {
144 uart_char = USART2->DR & 0xFF; // and read the data from peripheral
145 __HAL_UART_CLEAR_FLAG(&huart2, UART_FLAG_RXNE); // clear data available flag using HAL
146 }
147 return uart_char;
148}

151// void UART_gets
152// args: char *readbuffer
153// int size, size of readbuffer handed over
154// int echo, when TRUE, send read-char to UART
155// remark: ARM sends -1 if buffer is empty
156// LF is cleared if set in terminal-program
157void UART_gets(char *s, int size, int echo)
158{
159 int pos = 0;
160
161 while (1)
162 {
163 // ADD DELAY FOR SCHEDULER, IF TOO LONG, CHARS CAN BE MISSED...
164 osDelay(60);
165
166 *s = UART_get();
167
168 if (*s == -1 || *s == 255 || *s == CRETURN) // no data, or CR pressed
169 continue;
170
171 UART_putchar(*s); // show character
172
173 if (*s == LFEED) // check LF pressed
174 {
175 *s = '\0'; // now, close string
176 return;
177 }
178 else if (pos == size - 2) // check if last char reached
179 {
180 *++s = '\0'; // first, skip to last position
181 return;
182 }
183
184 s++;
185 pos++;
186 }
187}
